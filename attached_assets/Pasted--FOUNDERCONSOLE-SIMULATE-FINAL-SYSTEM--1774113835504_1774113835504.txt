import React, { useState, useEffect } from "react";

// ==========================================
// STATUS BAR
// ==========================================
function StatusBar() {
  return (
    <div className="flex items-center gap-2 text-sm text-gray-400 mb-3">
      <div className="status-dot"></div>
      <span>Running simulation...</span>
    </div>
  );
}

// ==========================================
// AGENT EVENT
// ==========================================
function AgentEvent({ event }: any) {
  const colorMap: any = {
    investor: "bg-purple-500/10 border-purple-500/30",
    customer: "bg-blue-500/10 border-blue-500/30",
    team: "bg-green-500/10 border-green-500/30",
    market: "bg-orange-500/10 border-orange-500/30",
  };

  return (
    <div className={`event-item flex gap-3 p-3 rounded-lg border ${colorMap[event.type]}`}>
      <div className="text-xs uppercase text-gray-400 w-16">
        {event.type}
      </div>

      <div className="flex-1 text-sm">
        <div>{event.message}</div>
        <div className="text-xs text-gray-500 mt-1">{event.time}</div>
      </div>
    </div>
  );
}

// ==========================================
// EVENT FEED
// ==========================================
function EventFeed({ events }: any) {
  return (
    <div className="space-y-2 max-h-[420px] overflow-y-auto pr-2">
      {events.map((e: any, i: number) => (
        <AgentEvent key={i} event={e} />
      ))}
    </div>
  );
}

// ==========================================
// TIMELINE
// ==========================================
function Timeline({ current }: { current: number }) {
  return (
    <div className="flex gap-2 mb-4">
      {[...Array(12)].map((_, i) => (
        <div
          key={i}
          className={`h-2 flex-1 rounded ${
            i <= current ? "bg-blue-500" : "bg-gray-700"
          }`}
        />
      ))}
    </div>
  );
}

// ==========================================

// ==========================================
// CUSTOMER COHORT VISUAL
// ==========================================

      <h3 className="text-sm mb-2">Customer Cohort</h3>

      <div className="space-y-2">

// ==========================================
// DECISION REPLAY (VIRAL FEATURE)
// ==========================================
function DecisionReplay({ events }: any) {
  const [index, setIndex] = useState(0);

  useEffect(() => {
    const i = setInterval(() => {
      setIndex((prev) => (prev + 1) % events.length);
    }, 1500);

    return () => clearInterval(i);
  }, []);

  return (
    <div className="result-card bg-white/5 border border-white/10 p-4 rounded-lg">
      <h3 className="text-sm mb-2">Decision Replay</h3>

      <div className="text-sm animate-fadeIn">
        {events[index]?.message}
      </div>

      <div className="text-xs text-gray-400 mt-2">
        Replay of simulation outcomes
      </div>
    </div>
  );
}

// ==========================================
// MAIN SIMULATION PAGE
// ==========================================
export default function SimulateEnhanced() {
  const [events, setEvents] = useState([
    { type: "team", message: "Hiring increased burn", time: "Month 2" },
    { type: "customer", message: "Churn increased", time: "Month 3" },
    { type: "investor", message: "Investor confidence dropped", time: "Month 5" },
    { type: "market", message: "Market slowdown detected", time: "Month 6" },
  ]);

  const runway = useCounter(14);
  const burn = useCounter(55000);
  const cash = useCounter(780000);

  const riskHigh = runway < 10;

  useEffect(() => {
    const interval = setInterval(() => {
      const newEvent = {
        type: ["investor", "customer", "team"][Math.floor(Math.random() * 3)],
        message: "Dynamic system update",
        time: `Month ${Math.floor(Math.random() * 12)}`,
      };

      setEvents((prev) => [newEvent, ...prev.slice(0, 8)]);
    }, 4000);

    return () => clearInterval(interval);
  }, []);

  return (
    <div className="simulation-root relative min-h-screen bg-[#0a0a0a] text-white p-6">

      <div className="bg-orb top-10 left-10"></div>
      <div className="bg-orb bottom-10 right-10"></div>

      <StatusBar />

      <div className="grid grid-cols-12 gap-6">

        {/* LEFT */}
        <div className="col-span-3 space-y-4">
          <div className="bg-white/5 border p-4 rounded-lg">
            <h3 className="text-sm mb-2">Live Metrics</h3>

            <div className="space-y-1 text-sm">
              <div>Cash: ${cash.toLocaleString()}</div>
              <div>Burn: ${burn.toLocaleString()}</div>
              <div>Runway: {runway} months</div>

              <div className={`mt-2 px-2 py-1 text-xs rounded ${riskHigh ? "risk-alert bg-red-500/20" : "bg-green-500/20"}`}>
                Risk: {riskHigh ? "High" : "Moderate"}
              </div>
            </div>
          </div>
        </div>

        {/* CENTER */}
        <div className="col-span-6">
          <Timeline current={6} />

          <div className="bg-white/5 border p-4 rounded-lg">
            <EventFeed events={events} />
          </div>
        </div>

        {/* RIGHT */}
        <div className="col-span-3 space-y-4">
          <InvestorPanel probability={62} />
          <CohortPanel />
          <DecisionReplay events={events} />
        </div>

      </div>

      <div className="mt-6 shimmer h-10 rounded flex items-center justify-center text-xs">
        Re-running simulation...
      </div>

    </div>
  );
}
