## MISSION

Redesign the entire FounderConsole UI/UX to a world-class, investor-demo-ready standard. The inspiration is mirofish-demo.pages.dev/console — a 5-stage simulation console — but we are going BEYOND it. We are not copying it. We are taking the best patterns and elevating them with better hierarchy, emotional pacing, micro-interactions, and polish.

This is a React 18 + TypeScript + Tailwind CSS app. The existing sidebar, routing, and data layer stay intact. We are upgrading the VISUAL LAYER and INTERACTION DESIGN only. Do NOT break existing functionality. All changes are additive or CSS/component-level refactors.

### 5. Create `client/src/components/ui/animated-counter.tsx`

An odometer-style animated number:
```tsx
import { useEffect, useRef, useState } from "react";

interface AnimatedCounterProps {
  value: number;
  duration?: number;
  prefix?: string;
  suffix?: string;
  className?: string;
  formatFn?: (n: number) => string;
}

export function AnimatedCounter({ value, duration = 1200, prefix, suffix, className, formatFn }: AnimatedCounterProps) {
  const [displayValue, setDisplayValue] = useState(0);
  const startRef = useRef(0);
  const startTimeRef = useRef<number | null>(null);

  useEffect(() => {
    startRef.current = displayValue;
    startTimeRef.current = null;

    const animate = (timestamp: number) => {
      if (!startTimeRef.current) startTimeRef.current = timestamp;
      const progress = Math.min((timestamp - startTimeRef.current) / duration, 1);
      const eased = 1 - Math.pow(1 - progress, 4); // ease-out-quart
      const current = Math.round(startRef.current + (value - startRef.current) * eased);
      setDisplayValue(current);
      if (progress < 1) requestAnimationFrame(animate);
    };

    requestAnimationFrame(animate);
  }, [value, duration]);

  const formatted = formatFn ? formatFn(displayValue) : displayValue.toLocaleString();

  return (
    <span className={className}>
      {prefix}{formatted}{suffix}
    </span>
  );
}
```

import { cn } from "@/lib/utils";
import { useState } from "react";
import { ChevronUp, ChevronDown, Terminal } from "lucide-react";

interface LogEntry {
  timestamp: string;
  message: string;
  type?: "info" | "success" | "error" | "warning";

  title?: string;
  projectId?: string;
  className?: string;

  return (

      expanded ? "h-64" : "h-10",
      className
    )}>
      {/* Collapsed bar */}

          <Terminal className="w-3.5 h-3.5" />
          <span className="text-[var(--text-tertiary)] uppercase tracking-wider">{title}</span>
          {lastLog && (
            <span className={cn(
              lastLog.type === "success" ? "text-[var(--accent-green)]" : 
              lastLog.type === "error" ? "text-[var(--accent-red)]" : "text-[var(--text-secondary)]"

                log.type === "error" ? "text-[var(--accent-red)]" :

### 15. Toast Notifications

Create `client/src/components/ui/event-toast.tsx` — a small notification that slides in from the bottom-right corner for simulation events:
```tsx
import { cn } from "@/lib/utils";
import { useEffect, useState } from "react";

interface EventToastProps {
  message: string;
  type?: "info" | "success" | "warning";
  duration?: number;
  onDismiss: () => void;
}

export function EventToast({ message, type = "info", duration = 4000, onDismiss }: EventToastProps) {
  const [visible, setVisible] = useState(false);

  useEffect(() => {
    requestAnimationFrame(() => setVisible(true));
    const timer = setTimeout(() => {
      setVisible(false);
      setTimeout(onDismiss, 300);
    }, duration);
    return () => clearTimeout(timer);
  }, []);

  return (
    <div className={cn(
      "fixed bottom-14 right-4 z-50 max-w-sm px-4 py-2.5 rounded-lg border font-mono text-xs transition-all duration-300",
      "bg-[var(--surface-elevated)] border-[var(--border-default)] shadow-[var(--shadow-md)]",
      visible ? "translate-y-0 opacity-100" : "translate-y-4 opacity-0",
      type === "success" && "border-[var(--accent-green)]/30",
      type === "warning" && "border-[var(--accent-amber)]/30",
    )}>
      <span className={cn(
        type === "success" ? "text-[var(--accent-green)]" :
        type === "warning" ? "text-[var(--accent-amber)]" :
        "text-[var(--text-secondary)]"
      )}>{type === "success" && "✓ "}{message}</span>
    </div>
  );
}
```
