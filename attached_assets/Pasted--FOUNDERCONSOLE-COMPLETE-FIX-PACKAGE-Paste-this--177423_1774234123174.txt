class CopilotService:
    """AI Copilot with retry logic and multiple LLM fallbacks"""
    
    def __init__(self):
        self.llm_configs = [
            {'name': 'openai', 'model': 'gpt-4', 'api_key': OPENAI_API_KEY},
            {'name': 'anthropic', 'model': 'claude-3-opus', 'api_key': ANTHROPIC_API_KEY},
        ]
        self.max_retries = 3
        self.timeout = 30
    
    def retry_with_backoff(self, func, max_retries=3):
        """Retry function with exponential backoff"""
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(max_retries):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    wait_time = (2 ** attempt) + (attempt * 0.1)
                    logger.warning(f"Attempt {attempt + 1} failed: {str(e)}. Retrying in {wait_time}s...")
                    time.sleep(wait_time)
                    if attempt == max_retries - 1:
                        raise e
            return None
        return wrapper
    
    def get_response(self, query, company_context):
        """Get AI response with fallback between providers"""
        
        system_prompt = f"""You are an expert financial advisor for startups. 
        Use this company data to answer: {company_context}
        Be concise, actionable, and data-driven. Always explain your reasoning."""
        
        # Try each LLM in order
        for llm_config in self.llm_configs:
            try:
                if llm_config['name'] == 'openai' and llm_config['api_key']:
                    response = self._call_openai(system_prompt, query)
                    return {
                        'response': response,
                        'model': 'gpt-4',
                        'confidence': 0.85,
                        'success': True
                    }
                    
                elif llm_config['name'] == 'anthropic' and llm_config['api_key']:
                    response = self._call_anthropic(system_prompt, query)
                    return {
                        'response': response,
                        'model': 'claude-3',
                        'confidence': 0.85,
                        'success': True
                    }
                    
            except Exception as e:
                logger.error(f"{llm_config['name']} failed: {str(e)}")
                continue
        
        # Fallback: Rule-based response
        return self._fallback_response(query, company_context)
    
    def _call_openai(self, system_prompt, user_query):
        """Call OpenAI API"""
        headers = {
            'Authorization': f'Bearer {OPENAI_API_KEY}',
            'Content-Type': 'application/json'
        }
        
        data = {
            'model': 'gpt-4',
            'messages': [
                {'role': 'system', 'content': system_prompt},
                {'role': 'user', 'content': user_query}
            ],
            'temperature': 0.7,
            'max_tokens': 500
        }
        
        response = requests.post(
            'https://api.openai.com/v1/chat/completions',
            headers=headers,
            json=data,
            timeout=self.timeout
        )
        
        if response.status_code == 200:
            return response.json()['choices'][0]['message']['content']
        else:
            raise Exception(f"OpenAI error: {response.status_code}")
    
    def _call_anthropic(self, system_prompt, user_query):
        """Call Anthropic API"""
        headers = {
            'x-api-key': ANTHROPIC_API_KEY,
            'Content-Type': 'application/json'
        }
        
        data = {
            'model': 'claude-3-opus-20240229',
            'max_tokens': 500,
            'system': system_prompt,
            'messages': [{'role': 'user', 'content': user_query}]
        }
        
        response = requests.post(
            'https://api.anthropic.com/v1/messages',
            headers=headers,
            json=data,
            timeout=self.timeout
        )
        
        if response.status_code == 200:
            return response.json()['content'][0]['text']
        else:
            raise Exception(f"Anthropic error: {response.status_code}")
    
    def _fallback_response(self, query, context):
        """Rule-based fallback when all LLMs fail"""
        query_lower = query.lower()
        
        if 'runway' in query_lower:
            return {
                'response': f"Based on your current burn rate of {context.get('burn_rate', 'N/A')} and cash of {context.get('cash', 'N/A')}, your runway is approximately {context.get('runway', 'N/A')} months. To extend runway, consider reducing burn by 15-20% or raising additional capital.",
                'model': 'fallback-rules',
                'confidence': 0.6,
                'success': True,
                'fallback': True
            }
        
        elif 'churn' in query_lower:
            return {
                'response': f"Your current churn rate is {context.get('churn_rate', 'N/A')}. Industry benchmark for SaaS is 5-7% annually. To reduce churn, focus on customer success, improve onboarding, and gather feedback from churned customers.",
                'model': 'fallback-rules',
                'confidence': 0.6,
                'success': True,
                'fallback': True
            }
        
        elif 'pricing' in query_lower:
            return {
                'response': "To evaluate pricing changes, consider: 1) Customer willingness to pay surveys, 2) Competitor pricing analysis, 3) Value-based pricing vs cost-plus. Test with a small cohort before full rollout.",
                'model': 'fallback-rules',
                'confidence': 0.5,
                'success': True,
                'fallback': True
            }
        
        else:
            return {
                'response': "I'm currently experiencing high demand. For immediate assistance, try asking about: runway, burn rate, churn, pricing, or fundraising timing. You can also run a simulation to test specific scenarios.",
                'model': 'fallback-generic',
                'confidence': 0.4,
                'success': True,
                'fallback': True
            }

class UnifiedDataService:
    """Single source of truth for ALL modules"""
    
    def __init__(self, company_id):
        self.company_id = company_id
        self._cache = {}
        self._cache_ttl = 60  # 60 seconds cache
    
    def get_all_metrics(self, force_refresh=False):
        """Get all metrics - used by Dashboard, Simulation, Digital Twin, Health Check"""
        cache_key = f"metrics_{self.company_id}"
        
        # Check cache
        if not force_refresh and cache_key in self._cache:
            cached = self._cache[cache_key]
            age = time.time() - cached['timestamp']
            if age < self._cache_ttl:
                logger.info(f"Returning cached metrics (age: {age:.1f}s)")
                return cached['data']
        
        # Fetch fresh data from sources
        metrics = {
            'mrr': self._fetch_mrr(),
            'arr': self._fetch_mrr() * 12,
            'burn_rate': self._fetch_burn_rate(),
            'cash_balance': self._fetch_cash(),
            'runway': self._calculate_runway(),
            'growth_rate': self._fetch_growth_rate(),
            'churn_rate': self._fetch_churn_rate(),
            'ltv': self._fetch_ltv(),
            'cac': self._fetch_cac(),
            'ltv_cac_ratio': self._fetch_ltv() / max(self._fetch_cac(), 1),
            'gross_margin': self._fetch_gross_margin(),
            'customers': self._fetch_customers(),
            'arpu': self._fetch_mrr() / max(self._fetch_customers(), 1),
            'payback_period': self._fetch_cac() / max(self._fetch_mrr() / max(self._fetch_customers(), 1), 1),
            'nrr': self._fetch_nrr(),
            'last_updated': datetime.utcnow().isoformat(),
            'data_sources': {
                'mrr': 'stripe',
                'burn_rate': 'quickbooks',
                'cash_balance': 'plaid',
                'churn_rate': 'stripe'
            }
        }
        
        # Cache result
        self._cache[cache_key] = {
            'data': metrics,
            'timestamp': time.time()
        }
        
        return metrics
    
    def _fetch_mrr(self):
        """Fetch from Stripe or database"""

        return 18500000  # $18.5M
    
    def _fetch_burn_rate(self):
        """Fetch from QuickBooks or database"""
        return 11400000  # $11.4M
    
    def _fetch_cash(self):
        """Fetch from Plaid or database"""
        return 35000000  # $35M
    
    def _fetch_growth_rate(self):
        """Calculate MoM growth"""
        return 0.10  # 10%
    
    def _fetch_churn_rate(self):
        """Fetch from Stripe"""
        return 0.032  # 3.2%
    
    def _fetch_ltv(self):
        """Calculate LTV"""
        return 500000  # $500K
    
    def _fetch_cac(self):
        """Fetch CAC"""
        return 50000  # $50K
    
    def _fetch_gross_margin(self):
        """Calculate gross margin"""
        return 0.824  # 82.4%
    
    def _fetch_customers(self):
        """Fetch customer count"""
        return 150
    
    def _fetch_nrr(self):
        """Fetch Net Revenue Retention"""
        return 1.08  # 108%
    
    def _calculate_runway(self):
        """Calculate runway in months"""
        cash = self._fetch_cash()
        burn = self._fetch_burn_rate()
        if burn <= 0:
            return 999  # Profitable
        return int(cash / burn)

import random
import numpy as np

@app.route('/api/simulation/run', methods=['POST'])
def run_simulation():
    """Run Monte Carlo simulation"""
    try:
        data = request.json or {}
        company_id = data.get('company_id', 'default')
        scenario = data.get('scenario', {})
        
        # Get baseline metrics
        metrics = get_company_metrics(company_id)
        
        # Apply scenario changes
        growth_change = scenario.get('growth_change', 0)
        burn_change = scenario.get('burn_change', 0)
        churn_change = scenario.get('churn_change', 0)
        
        # Run Monte Carlo (simplified)
        iterations = 1000
        results = []
        
        for _ in range(iterations):
            # Add randomness
            growth = metrics['growth_rate'] * (1 + growth_change) * random.uniform(0.8, 1.2)
            burn = metrics['burn_rate'] * (1 + burn_change) * random.uniform(0.9, 1.1)
            churn = metrics['churn_rate'] * (1 + churn_change) * random.uniform(0.9, 1.1)
            
            # Calculate runway
            if burn <= 0:
                runway = 999
            else:
                runway = metrics['cash_balance'] / burn
            
            results.append({
                'runway': runway,
                'revenue': metrics['mrr'] * (1 + growth) ** 18,
                'survival': 1 if runway > 18 else 0
            })
        
        # Calculate percentiles
        runways = [r['runway'] for r in results]
        revenues = [r['revenue'] for r in results]
        
        p10 = np.percentile(runways, 10)
        p50 = np.percentile(runways, 50)
        p90 = np.percentile(runways, 90)
        
        survival_rate = sum(r['survival'] for r in results) / len(results) * 100
        
        return jsonify({
            'success': True,
            'scenario_name': scenario.get('name', 'Custom Scenario'),
            'iterations': iterations,
            'results': {
                'runway_p10': round(p10, 1),
                'runway_p50': round(p50, 1),
                'runway_p90': round(p90, 1),
                'revenue_p50': round(np.percentile(revenues, 50) / 1000000, 1),
                'survival_18m': round(survival_rate, 1),
                'confidence_spread': round(p90 - p10, 1)
            },
            'decision_score': 8 if survival_rate > 90 else 6 if survival_rate > 70 else 4,
            'verdict': 'GO' if survival_rate > 90 else 'CONDITIONAL' if survival_rate > 70 else 'NO-GO',
            'timestamp': datetime.utcnow().isoformat()
        })
        
    except Exception as e:
        logger.error(f"Simulation error: {str(e)}")
        return jsonify({'success': False, 'error': str(e)}), 500
